-- ============================================================
local HttpService = game:GetService("HttpService")
local SAVE_FILE   = "DeltaFarmV28_Settings.json"
local SaveData    = {}
local SaveDirty   = false
local SaveReady   = false

pcall(function()
    if isfile and isfile(SAVE_FILE) then
        local raw = readfile(SAVE_FILE)
        if raw and #raw > 0 then
            local ok, decoded = pcall(function()
                return HttpService:JSONDecode(raw)
            end)
            if ok and type(decoded) == "table" then
                SaveData = decoded
            end
        end
    end
    SaveReady = true
end)

local SaveWriting = false
local function SaveWrite()
    if SaveWriting or not SaveDirty or not writefile then return end
    SaveWriting = true
    local ok = pcall(function()
        writefile(SAVE_FILE, HttpService:JSONEncode(SaveData))
    end)
    if ok then
        SaveDirty = false
    end
    SaveWriting = false
end

task.spawn(function()
    while task.wait(2) do
        SaveWrite()
    end
end)

local function SaveSet(key, value)
    SaveData[key] = value
    SaveDirty = true
end

local function SaveGet(key, default)
    local v = SaveData[key]
    if v == nil then return default end
    return v
end

local function RefreshCharacterCache(char)
    char = char or player.Character
    if RuntimeCache.Character == char and RuntimeCache.CharacterPartsReady then
        return char, RuntimeCache.HRP, RuntimeCache.Humanoid
    end

    RuntimeCache.Character = char
    RuntimeCache.HRP = char and char:FindFirstChild("HumanoidRootPart") or nil
    RuntimeCache.Humanoid = char and char:FindFirstChildOfClass("Humanoid") or nil
    table.clear(RuntimeCache.CharacterParts)
    table.clear(RuntimeCache.CharacterPartSet)
    RuntimeCache.CharacterPartsReady = false

    if char then
        for _, d in ipairs(char:GetDescendants()) do
            if d:IsA("BasePart") then
                RuntimeCache.CharacterParts[#RuntimeCache.CharacterParts + 1] = d
                RuntimeCache.CharacterPartSet[d] = true
            end
        end
        RuntimeCache.CharacterPartsReady = true
    end
    return char, RuntimeCache.HRP, RuntimeCache.Humanoid
end

local function GetCachedCharacter()
    local char = player.Character
    if RuntimeCache.Character ~= char then
        return RefreshCharacterCache(char)
    end
    if char and (not RuntimeCache.HRP or not RuntimeCache.HRP.Parent) then
        RuntimeCache.HRP = char:FindFirstChild("HumanoidRootPart")
    end
    if char and (not RuntimeCache.Humanoid or not RuntimeCache.Humanoid.Parent) then
        RuntimeCache.Humanoid = char:FindFirstChildOfClass("Humanoid")
    end
    return char, RuntimeCache.HRP, RuntimeCache.Humanoid
end

        if d:IsA("BasePart") and not RuntimeCache.CharacterPartSet[d] then

        if d:IsA("Humanoid") then
            RuntimeCache.Humanoid = d
        end
    end)
end)

-- ============================================================
-- 🌫️ NOCLIP
-- ============================================================
local function ApplyNoclip()
    local char = GetCachedCharacter()
    if not char then return end
    pcall(function()
        for i = 1, #RuntimeCache.CharacterParts do
            local part = RuntimeCache.CharacterParts[i]
            if part and part.Parent and part.CanCollide then
                part.CanCollide = false
            end
        end
    end)
end

local function RemoveNoclip()
    local char = GetCachedCharacter()
    if not char then return end
    pcall(function()
        local HRP = RuntimeCache.HRP
        if HRP then HRP.CanCollide = true end
        local torso = char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
        if torso then torso.CanCollide = true end
        local head = char:FindFirstChild("Head")
        if head then head.CanCollide = true end
    end)
end

-- ============================================================
-- 🧊 FREEZE PLAYER
-- ============================================================
local function FreezePlayer(state)
    local char = player.Character
    local HRP = char and char:FindFirstChild("HumanoidRootPart")
    if not HRP then return end
    if state then
        if not HRP:FindFirstChild("ChestFreeze") then
            local bv = Instance.new("BodyVelocity")
            bv.Name = "ChestFreeze"
            bv.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
            bv.Velocity = Vector3.new(0, 0, 0)
            bv.Parent = HRP
        end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = true end
    else
        local old = HRP:FindFirstChild("ChestFreeze")
        if old then old:Destroy() end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = false end
    end
end

-- 🚀 FAST ATTACK ENGINE V2
-- ============================================================
local FastAttackModule = (function()
    local function SafeWaitForChild(parent, childName, timeout)
        local ok, result = pcall(function()
            return parent:WaitForChild(childName, timeout or 5)
        end)
        return ok and result or nil
    end

    local Remotes = SafeWaitForChild(ReplicatedStorage, "Remotes", 5)
    if not Remotes then warn("⚠️ [FastAttack] Remotes not found"); return nil end

    local Modules = SafeWaitForChild(ReplicatedStorage, "Modules", 5)
    local Net = Modules and SafeWaitForChild(Modules, "Net", 5)
    if not Net then warn("⚠️ [FastAttack] Net module not found"); return nil end

    local RegisterAttack = SafeWaitForChild(Net, "RE/RegisterAttack", 5)
    local RegisterHit    = SafeWaitForChild(Net, "RE/RegisterHit", 5)
    if not RegisterAttack or not RegisterHit then
        warn("⚠️ [FastAttack] RegisterAttack/Hit not found"); return nil
    end

    local Workspace  = game:GetService("Workspace")
    local Enemies    = Workspace:FindFirstChild("Enemies")
    local Characters = Workspace:FindFirstChild("Characters")

    local FolderCaches = {}
    local function AttachFolderCache(folder)
        if not folder then return nil end
        local existing = FolderCaches[folder]
        if existing then return existing.items end

        local cache = {items = {}, index = {}, data = {}, connections = {}}
        FolderCaches[folder] = cache

        local function cacheMob(child)
            if not child or cache.index[child] then return end
            cache.items[#cache.items + 1] = child
            cache.index[child] = #cache.items
            cache.data[child] = {
                head = child:FindFirstChild("Head"),
                hrp = child:FindFirstChild("HumanoidRootPart"),
                humanoid = child:FindFirstChildOfClass("Humanoid"),
            }
        end

        for _, child in ipairs(folder:GetChildren()) do
            cacheMob(child)
        end

        cache.connections[#cache.connections + 1] = folder.ChildAdded:Connect(function(child)
            cacheMob(child)
        end)
        cache.connections[#cache.connections + 1] = folder.ChildRemoved:Connect(function(child)
            local index = cache.index[child]
            if not index then return end
            local last = cache.items[#cache.items]
            cache.items[index] = last
            cache.index[last] = index
            cache.items[#cache.items] = nil
            cache.index[child] = nil
            cache.data[child] = nil
        end)

        return cache.items
    end

    local function DetachFolderCache(folder)
        local cache = folder and FolderCaches[folder]
        if not cache then return end
        for _, connection in ipairs(cache.connections) do
            pcall(function() connection:Disconnect() end)
        end
        FolderCaches[folder] = nil
    end

    if Enemies then AttachFolderCache(Enemies) end
    if Characters then AttachFolderCache(Characters) end

    Workspace.ChildAdded:Connect(function(child)
        if child.Name == "Enemies" then
            if Enemies and Enemies ~= child then DetachFolderCache(Enemies) end
            Enemies = child
            AttachFolderCache(Enemies)
        elseif child.Name == "Characters" then
            if Characters and Characters ~= child then DetachFolderCache(Characters) end
            Characters = child
            AttachFolderCache(Characters)
        end
    end)

    Workspace.ChildRemoved:Connect(function(child)
        if child == Enemies then
            DetachFolderCache(Enemies)
            Enemies = nil
        elseif child == Characters then
            DetachFolderCache(Characters)
            Characters = nil
        end
    end)

    local MIN_TICK        = 0.03
    local ATTACK_COOLDOWN = 0.125

    local FastAttack = {
        Distance       = 500,
        TargetName     = nil,
        attackMobs     = true,
        attackPlayers  = false,
    }

    local lastAttackTime = 0
    local cachedWeapon = nil
    local cachedWeaponName = nil
    local cachedLeftClickRemote = nil

    local function GetEquippedWeapon(char)
        if not char then return nil end
        local tool = char:FindFirstChildOfClass("Tool")
        if not tool or not tool.Parent then return nil end
        return tool
    end

    local function RefreshWeapon(char)
        local tool = GetEquippedWeapon(char)
        if tool ~= cachedWeapon or (tool and tool.Name ~= cachedWeaponName) then
            cachedWeapon = tool
            cachedWeaponName = tool and tool.Name or nil
            cachedLeftClickRemote = tool and tool:FindFirstChild("LeftClickRemote") or nil
        end
        return cachedWeapon
    end

    local function MarkMobs(list)
        local now = tick()
        for _, e in ipairs(list) do
            local mob = e[1]
            if mob and mob.Parent then
                MarkedMobs[mob] = now
            end
        end
    end

    local function CollectMobs(folder, targetName, playerPos, limitDist, result, horizontalOnly)
        result = result or {}
        table.clear(result)
        local nearest, nearestDistSq = nil, math.huge
        if not folder then return result, nil end

        local limitDistSq = limitDist * limitDist
        local localCharacter = LocalPlayer.Character
        local children = AttachFolderCache(folder)
        local folderCache = FolderCaches[folder]
        local data = folderCache and folderCache.data
        for i = 1, #children do
            local enemy = children[i]
            if enemy ~= localCharacter
               and ((not targetName) or enemy.Name == targetName) then
                local entry = data and data[enemy]
                local head = entry and entry.head
                local hum = entry and entry.humanoid

                if (not head or not head.Parent) or (not hum or not hum.Parent) then
                    head = enemy:FindFirstChild("Head")
                    hum = enemy:FindFirstChildOfClass("Humanoid")
                    if entry then
                        entry.head = head
                        entry.humanoid = hum
                        entry.hrp = enemy:FindFirstChild("HumanoidRootPart")
                    end
                end

                if head and hum and hum.Health > 0 then
                    local hp = head.Position
                    local dx = hp.X - playerPos.X
                    local dy = hp.Y - playerPos.Y
                    local dz = hp.Z - playerPos.Z
                    local dSq
                    if horizontalOnly then
                        dSq = dx * dx + dz * dz
                    else
                        dSq = dx * dx + dy * dy + dz * dz
                    end
                    if dSq < limitDistSq then
                        result[#result + 1] = {enemy, head}
                        if dSq < nearestDistSq then
                            nearestDistSq, nearest = dSq, head
                        end
                    end
                end
            end
        end
        return result, nearest
    end

    function FastAttack:Attack(BasePart, list)
        if not BasePart or #list == 0 then return end
        local now = tick()
        if now - lastAttackTime < ATTACK_COOLDOWN then return end
        lastAttackTime = now
        MarkMobs(list)

        pcall(function()
            RegisterAttack:FireServer(0)
            RegisterHit:FireServer(BasePart, list)
        end)
    end

    local mobsECache, mobsCCache, allMobsCache = {}, {}, {}

    function FastAttack:AttackNearest(targetName)
        local char = LocalPlayer.Character
        if not char then return end

        local HRP = char:FindFirstChild("HumanoidRootPart")
        if not HRP then return end

        local playerPos = HRP.Position
        local mobsE, nearestE = CollectMobs(Enemies, targetName, playerPos, self.Distance, mobsECache)
        local mobsC, nearestC = CollectMobs(Characters, targetName, playerPos, self.Distance, mobsCCache, true)

        table.clear(allMobsCache)
        for i = 1, #mobsE do allMobsCache[#allMobsCache + 1] = mobsE[i] end
        for i = 1, #mobsC do allMobsCache[#allMobsCache + 1] = mobsC[i] end
        local allMobs = allMobsCache

        if #allMobs == 0 then
            return
        end

        local weapon = RefreshWeapon(char)
        if not weapon then return end

        local tip = weapon.ToolTip or ""

        if tip == "Gun" then
            return
        end

        local leftClick = cachedLeftClickRemote
        if not leftClick or not leftClick.Parent then
            leftClick = weapon:FindFirstChild("LeftClickRemote")
            cachedLeftClickRemote = leftClick
        end

        if tip == "Blox Fruit" then
            if tick() - lastAttackTime < ATTACK_COOLDOWN then return end
            lastAttackTime = tick()
            MarkMobs(allMobs)

            if leftClick and leftClick:IsA("RemoteEvent") then
                for _, enemyData in ipairs(allMobs) do
                    local enemy = enemyData[1]
                    local eHRP = enemy and enemy:FindFirstChild("HumanoidRootPart")
                    if eHRP then
                        local delta = eHRP.Position - playerPos
                        if delta.Magnitude > 0 then
                            pcall(function()
                                leftClick:FireServer(delta.Unit, 1)
                            end)
                        end
                    end
                end
            else
                pcall(function()
                    if weapon.Parent == char then
                        weapon:Activate()
                    end
                end)
            end
            return
        end

        if leftClick and leftClick:IsA("RemoteEvent") then
            if tick() - lastAttackTime < ATTACK_COOLDOWN then return end
            lastAttackTime = tick()
            MarkMobs(allMobs)

            for _, enemyData in ipairs(allMobs) do
                local enemy = enemyData[1]
                local eHRP = enemy and enemy:FindFirstChild("HumanoidRootPart")
                if eHRP then
                    local delta = eHRP.Position - playerPos
                    if delta.Magnitude > 0 then
                        pcall(function()
                            leftClick:FireServer(delta.Unit, 1)
                        end)
                    end
                end
            end
            return
        end

        local best = nearestE or nearestC
        if best and (tip == "Melee" or tip == "Sword" or tip == "") then
            self:Attack(best, allMobs)
        end
    end

    function FastAttack:BladeHits(targetName)
        self:AttackNearest(targetName)
    end

    task.spawn(function()
        while task.wait(MIN_TICK) do
            if _G.AutoFarm then
                pcall(function()
                    FastAttack:BladeHits(FastAttack.TargetName)
                end)
            end
        end
    end)

    _G.LocalFastAttack = (_G.LocalFastAttack ~= false)
    task.spawn(function()
        while task.wait(MIN_TICK) do
            if not _G.AutoFarm and _G.LocalFastAttack then
                pcall(function()
                    FastAttack:BladeHits(nil)
                end)
            end
        end
    end)

    task.spawn(function()
        while task.wait(3) do
            if not Enemies or not Enemies.Parent then
                local folder = Workspace:FindFirstChild("Enemies")
                if folder ~= Enemies then
                    if Enemies then DetachFolderCache(Enemies) end
                    Enemies = folder
                    if Enemies then AttachFolderCache(Enemies) end
                end
            end
            if not Characters or not Characters.Parent then
                local folder = Workspace:FindFirstChild("Characters")
                if folder ~= Characters then
                    if Characters then DetachFolderCache(Characters) end
                    Characters = folder
                    if Characters then AttachFolderCache(Characters) end
                end
            end
        end
    end)

    return FastAttack
end)()

local function MakeDraggable(guiObject, handleObject)
    handleObject = handleObject or guiObject
    local dragging, dragStart, startPos, dragConn
    handleObject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            dragging  = true
            dragStart = input.Position
            startPos  = guiObject.Position
            if dragConn then dragConn:Disconnect() end
            dragConn = input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                    if dragConn then dragConn:Disconnect() dragConn = nil end
                end
            end)
        end
    end)
    handleObject.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            local delta = input.Position - dragStart
            guiObject.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)
end

        end
    end)

end

local currentTween = nil
local function topos(Pos)
    if not Pos then return end
    local char = player.Character
    local HRP = char and char:FindFirstChild("HumanoidRootPart")
    if not HRP then return end
    if currentTween then pcall(function() currentTween:Cancel() end) end

    local startPos = HRP.Position
    local endPos = Pos.Position
    local speed = math.clamp(_G.FarmSpeed or 100, 1, 100) * 5

    local dx = endPos.X - startPos.X
    local dz = endPos.Z - startPos.Z
    local horizDistSq = dx * dx + dz * dz

    if horizDistSq > 900 then
        local travelY = math.max(startPos.Y, endPos.Y)
        local targetPos = CFrame.new(endPos.X, travelY, endPos.Z)
        local target = targetPos.Position
        local tx, ty, tz = target.X - startPos.X, target.Y - startPos.Y, target.Z - startPos.Z
        local dist = math.sqrt(tx * tx + ty * ty + tz * tz)
        local duration = dist / speed
        currentTween = TweenService:Create(HRP, TweenInfo.new(duration, Enum.EasingStyle.Linear), {CFrame = targetPos})
        currentTween:Play()
    else
        local dy = endPos.Y - startPos.Y
        local dist = math.sqrt(dx * dx + dy * dy + dz * dz)
        local duration = math.max(dist / speed, 0.05)
        currentTween = TweenService:Create(HRP, TweenInfo.new(duration, Enum.EasingStyle.Linear), {CFrame = Pos})
        currentTween:Play()
    end
end
function TP1(Pos) topos(Pos) end

local QuestLookupCache = {}
local function FindQuestByLevel(level)
    local playerTeam = GetPlayerTeam()
    local currentWorld = GetCurrentWorld()
    local cacheKey = tostring(currentWorld) .. "|" .. tostring(playerTeam) .. "|" .. tostring(level)
    local cached = QuestLookupCache[cacheKey]
    if cached ~= nil then
        return cached or nil
    end

    local bestQuest = nil
    local highestLevel = -1
    for _, q in ipairs(QUEST_DATABASE) do
        if q.World ~= currentWorld then continue end
        if q.Team and q.Team ~= playerTeam then continue end
        if level >= q.Level and q.Level > highestLevel then
            highestLevel = q.Level
            bestQuest = q
        end
    end
    QuestLookupCache[cacheKey] = bestQuest or false
    return bestQuest
end

local mobConnections = {}
local WatchedMobs = {}

local function DisconnectAllMobConnections()
    for _, conn in ipairs(mobConnections) do
        pcall(function() conn:Disconnect() end)
    end
    mobConnections = {}
    WatchedMobs = {}
end

local function PrintProgress()
    local curr = QuestData.Progress
    local req  = QuestData.Required
    if req == 0 then return end
    local ratio = curr / req
    local filled = math.floor(ratio * 20)
    local bar = string.rep("█", filled) .. string.rep("░", 20 - filled)
    print(string.format("   📊 [%s] %d/%d  [%s]  %d%%",
        QuestData.TargetMobName or "?",
        curr, req, bar, math.floor(ratio * 100)))
end

local function WatchMob(mob)
    if not mob or not mob:FindFirstChildOfClass("Humanoid") then return end
    if WatchedMobs[mob] then return end
    WatchedMobs[mob] = true

    local humanoid = mob:FindFirstChildOfClass("Humanoid")
    local counted = false

    local function OnDeath()
        if counted then return end
        if not QuestData.TargetMobName then return end
        if mob.Name ~= QuestData.TargetMobName then return end
        counted = true

        local markTime = MarkedMobs[mob]
        if not markTime or (tick() - markTime) > MARK_DURATION then
            return
        end

        QuestData.Progress = QuestData.Progress + 1
        QuestData.LastKillTime = tick()

        print(string.format("💀 [قتل #%d] %s  →  %d/%d",
            QuestData.Progress, mob.Name,
            QuestData.Progress, QuestData.Required))

        PrintProgress()

        if QuestData.Progress >= QuestData.Required then
            print("🎉 ═══ اكتمل العدد المطلوب! ═══")
        end
    end

    local diedConn = humanoid.Died:Connect(OnDeath)
    table.insert(mobConnections, diedConn)

    local removingConn = mob.Destroying:Connect(function()
        local h = mob:FindFirstChildOfClass("Humanoid")
        if h and h.Health <= 0 then OnDeath() end
    end)
    table.insert(mobConnections, removingConn)

    local healthConn = humanoid:GetPropertyChangedSignal("Health"):Connect(function()
        if humanoid.Health <= 0 then OnDeath() end
    end)
    table.insert(mobConnections, healthConn)
end

local function StartKillTracking()
    DisconnectAllMobConnections()
    local enemies = workspace:FindFirstChild("Enemies")
    if not enemies then return end

    for _, mob in ipairs(enemies:GetChildren()) do
        WatchMob(mob)
    end

    local addedConn = enemies.ChildAdded:Connect(function(mob)
        task.wait(0.1)
        WatchMob(mob)
    end)
    table.insert(mobConnections, addedConn)
end

local CHEST_GROUP_RADIUS = 2500
local STUCK_TIMEOUT      = 12

local ChestGroups = {}
local CurrentGroupIndex = 1
local LastChestTime = 0
local IsResetting = false

local ChestTagCache = {}
local ChestTagSet = {}
local ChestTagIndex = {}
for _, chest in ipairs(CollectionService:GetTagged("_ChestTagged")) do
    ChestTagCache[#ChestTagCache + 1] = chest
    ChestTagSet[chest] = true
    ChestTagIndex[chest] = #ChestTagCache
end
CollectionService:GetInstanceAddedSignal("_ChestTagged"):Connect(function(chest)
    if not ChestTagSet[chest] then
        ChestTagSet[chest] = true
        ChestTagCache[#ChestTagCache + 1] = chest
        ChestTagIndex[chest] = #ChestTagCache
    end
end)
CollectionService:GetInstanceRemovedSignal("_ChestTagged"):Connect(function(chest)
    ChestTagSet[chest] = nil
    local index = ChestTagIndex[chest]
    if index then
        local last = ChestTagCache[#ChestTagCache]
        ChestTagCache[index] = last
        if last then ChestTagIndex[last] = index end
        ChestTagCache[#ChestTagCache] = nil
        ChestTagIndex[chest] = nil
    end
end)

local function GetChestSnapshot()
    return ChestTagCache
end

local function GroupChests()
    local chests = GetChestSnapshot()
    local groups = {}
    local used = {}
    local buckets = {}
    local cellSize = CHEST_GROUP_RADIUS

    local function bucketKey(pos)
        return math.floor(pos.X / cellSize) .. ":" .. math.floor(pos.Y / cellSize) .. ":" .. math.floor(pos.Z / cellSize)
    end

    local positions = {}
    for i = 1, #chests do
        local chest = chests[i]
        if chest and ChestTagSet[chest] and chest:IsDescendantOf(workspace)
           and not chest:GetAttribute("IsDisabled")
           and MatchesChestFilter(chest) then
            local pos
            pcall(function() pos = chest:GetPivot().Position end)
            if pos then
                positions[chest] = pos
                local key = bucketKey(pos)
                local bucket = buckets[key]
                if not bucket then bucket = {}; buckets[key] = bucket end
                bucket[#bucket + 1] = chest
            end
        end
    end

    for i = 1, #chests do
        local chest = chests[i]
        local pos = positions[chest]
        if pos and not used[chest] then
            local group = {chest}
            used[chest] = true
            local bx, by, bz = math.floor(pos.X / cellSize), math.floor(pos.Y / cellSize), math.floor(pos.Z / cellSize)
            for ox = -1, 1 do
                for oy = -1, 1 do
                    for oz = -1, 1 do
                        local bucket = buckets[(bx + ox) .. ":" .. (by + oy) .. ":" .. (bz + oz)]
                        if bucket then
                            for j = 1, #bucket do
                                local other = bucket[j]
                                if other ~= chest and not used[other] then
                                    local otherPos = positions[other]
                                    if otherPos then
                                        local dx = otherPos.X - pos.X
                                        local dy = otherPos.Y - pos.Y
                                        local dz = otherPos.Z - pos.Z
                                        if dx * dx + dy * dy + dz * dz < CHEST_GROUP_RADIUS * CHEST_GROUP_RADIUS then
                                            group[#group + 1] = other
                                            used[other] = true
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
            groups[#groups + 1] = group
        end
    end
    return groups
end

local function GetGroupCenter(group)
    if #group == 0 then return nil end
    local sum = Vector3.new(0, 0, 0)
    local count = 0
    for _, chest in ipairs(group) do
        local pos
        pcall(function() pos = chest:GetPivot().Position end)
        if pos then
            sum = sum + pos
            count = count + 1
        end
    end
    if count == 0 then return nil end
    return sum / count
end

local function ResetCharacter()
    if IsResetting then return end
    IsResetting = true

    pcall(function()
        local char = player.Character
        if char then
            local HRP = char:FindFirstChild("HumanoidRootPart")
            if HRP then
                local freeze = HRP:FindFirstChild("ChestFreeze")
                if freeze then freeze:Destroy() end
            end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                hum.PlatformStand = false
                hum.Health = 0
            end
        end
    end)

    task.wait(4)
    IsResetting = false
end

task.spawn(function()
    while task.wait(0.05) do
        if _G.FarmChest then
            if IsResetting then continue end

            pcall(function() FreezePlayer(true) end)
            pcall(function() ApplyNoclip() end)

            pcall(function()
                local Character = player.Character
                if not Character then return end
                local HRP = Character:FindFirstChild("HumanoidRootPart")
                if not HRP then return end

                local Position = HRP.Position

                if _G.ChestIslandMode then

                    if #ChestGroups == 0 or CurrentGroupIndex > #ChestGroups then
                        ChestGroups = GroupChests()
                        CurrentGroupIndex = 1
                        LastChestTime = tick()
                        print(string.format("🏝️ تم إنشاء %d جزيرة", #ChestGroups))
                    end

                    if CurrentGroupIndex <= #ChestGroups then
                        local group = ChestGroups[CurrentGroupIndex]

                        local cleaned = {}
                        for i = 1, #group do
                            local c = group[i]
                            if c and ChestTagSet[c] and c:IsDescendantOf(workspace)
                               and not c:GetAttribute("IsDisabled") then
                                cleaned[#cleaned + 1] = c
                            end
                        end
                        ChestGroups[CurrentGroupIndex] = cleaned

                        if #cleaned == 0 then
                            print(string.format("✅ جزيرة %d خلصت → Reset & انتقل للجزيرة %d",
                                CurrentGroupIndex, CurrentGroupIndex + 1))

                            local nextIndex = CurrentGroupIndex + 1

                            if nextIndex <= #ChestGroups then
                                local nextCenter = GetGroupCenter(ChestGroups[nextIndex])
                                if nextCenter then
                                    pcall(function()
                                        HRP.CFrame = CFrame.new(nextCenter + Vector3.new(0, 50, 0))
                                    end)
                                    task.wait(0.5)
                                end
                            end

                            ResetCharacter()

                            CurrentGroupIndex = nextIndex
                            LastChestTime = tick()

                            if CurrentGroupIndex > #ChestGroups then
                                print("🎉 كل الجزر خلصت → جولة جديدة")
                                task.wait(2)
                                ChestGroups = GroupChests()
                                CurrentGroupIndex = 1
                                LastChestTime = tick()
                            end
                        else
                            local nearest, minDist = nil, math.huge
                            for _, c in ipairs(cleaned) do
                                local cPos
                                pcall(function() cPos = c:GetPivot().Position end)
                                if cPos then
                                    local d = (cPos - Position).Magnitude
                                    if d < minDist then
                                        minDist = d
                                        nearest = c
                                    end
                                end
                            end

                            if nearest then
                                local cPos = nearest:GetPivot().Position
                                toposChest(CFrame.new(cPos))

                                if minDist < 15 then
                                    LastChestTime = tick()
                                end
                            end

                            if tick() - LastChestTime > STUCK_TIMEOUT then
                                print(string.format("⚠️ عالق في جزيرة %d → Reset", CurrentGroupIndex))

                                local nextIndex = CurrentGroupIndex + 1
                                if nextIndex <= #ChestGroups then
                                    local nextCenter = GetGroupCenter(ChestGroups[nextIndex])
                                    if nextCenter then
                                        pcall(function()
                                            HRP.CFrame = CFrame.new(nextCenter + Vector3.new(0, 50, 0))
                                        end)
                                        task.wait(0.5)
                                    end
                                end

                                ResetCharacter()

                                CurrentGroupIndex = nextIndex
                                LastChestTime = tick()

                                if CurrentGroupIndex > #ChestGroups then
                                    ChestGroups = GroupChests()
                                    CurrentGroupIndex = 1
                                end
                            end
                        end
                    end

                else
                    local Chests = GetChestSnapshot()
                    local DistanceSq, Nearest = math.huge, nil

                    for i = 1, #Chests do
                        local Chest = Chests[i]
                        if Chest and ChestTagSet[Chest]
                           and Chest:IsDescendantOf(workspace)
                           and not Chest:GetAttribute("IsDisabled")
                           and MatchesChestFilter(Chest) then
                            local ChestPos = Chest:GetPivot().Position
                            local dx = ChestPos.X - Position.X
                            local dy = ChestPos.Y - Position.Y
                            local dz = ChestPos.Z - Position.Z
                            local dSq = dx * dx + dy * dy + dz * dz
                            if dSq < DistanceSq then
                                DistanceSq, Nearest = dSq, Chest
                            end
                        end
                    end

                    if Nearest then
                        local ChestPosition = Nearest:GetPivot().Position
                        toposChest(CFrame.new(ChestPosition))
                    end
                end
            end)
        else
            pcall(function() FreezePlayer(false) end)
            pcall(function() RemoveNoclip() end)
            ChestGroups = {}
            CurrentGroupIndex = 1
        end
    end
end)

    local V4PlayerTrialCorners = {
        Vector3.new(28769.60, 14890.70, -244.76),
        Vector3.new(28866.21, 14891.40, -123.32),
        Vector3.new(28697.89, 14889.70, 70.31),
        Vector3.new(28600.99, 14889.70, -81.58),
    }

    local function PointInV4TrialArena(position)
        if not position then return false end
        if position.Y < 14870 or position.Y > 14930 then return false end

        local inside = false
        local j = #V4PlayerTrialCorners
        for i = 1, #V4PlayerTrialCorners do
            local a = V4PlayerTrialCorners[i]
            local b = V4PlayerTrialCorners[j]
            local crosses = ((a.Z > position.Z) ~= (b.Z > position.Z))
            if crosses then
                local xAtZ = (b.X - a.X) * (position.Z - a.Z) / (b.Z - a.Z) + a.X
                if position.X < xAtZ then inside = not inside end
            end
            j = i
        end
        return inside
    end

    local function IsInsideTrialPosition(position, extra)
        if PointInV4TrialArena(position) then return true end
        if extra == nil then return false end
        local loc = GetTrialLocation()
        if not loc then return false end
        local pivot = loc:GetPivot().Position
        return (position - pivot).Magnitude <= extra
    end

    local function FindNearestTrialEnemy()
        local char = player.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        local loc = GetTrialLocation()
        if not hrp or not loc then return nil end

        local _, race = GetPlayerRaceV4()

        if race == "Shark" then
            local seaBeasts = workspace:FindFirstChild("SeaBeasts")
            local beast = seaBeasts and seaBeasts:FindFirstChild("SeaBeast1")
            local hum = beast and beast:FindFirstChildOfClass("Humanoid")
            local root = beast and beast:FindFirstChild("HumanoidRootPart")
            if hum and root and hum.Health > 0 then
                return beast, root
            end
        end

        local enemies = workspace:FindFirstChild("Enemies")
        if not enemies then return nil end

        local best, bestRoot, bestDist = nil, nil, math.huge
        local center = loc:GetPivot().Position
        for _, mob in ipairs(enemies:GetChildren()) do
            local hum = mob:FindFirstChildOfClass("Humanoid")
            local root = mob:FindFirstChild("HumanoidRootPart")
            if hum and root and hum.Health > 0 then
                local inTrial = (root.Position - center).Magnitude <= 350
                local d = (root.Position - hrp.Position).Magnitude
                if inTrial and d < bestDist and d <= 300 then
                    best, bestRoot, bestDist = mob, root, d
                end
            end
        end
        return best, bestRoot
    end

    local function AttackTrialEnemy()
        if not FastAttackModule then return false end
        local mob, root = FindNearestTrialEnemy()
        if not mob or not root then return false end
        EquipWeapon(GetSelectedWeapon and GetSelectedWeapon() or _G.WeaponType or "Melee")
        AutoHaki()
        local head = mob:FindFirstChild("Head") or root
        pcall(function() FastAttackModule:Attack(head, {{mob, head}}) end)
        return true
    end

    local function FindTrialParticipant()
        local my = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if not my then return nil end
        local best, bestRoot, bestHum, bestDist = nil, nil, nil, math.huge
        for _, plr in ipairs(game:GetService("Players"):GetPlayers()) do
            if plr ~= player and plr.Character then
                local ch = plr.Character
                local root = ch:FindFirstChild("HumanoidRootPart")
                local hum = ch:FindFirstChildOfClass("Humanoid")
                if root and hum and hum.Health > 0 and IsInsideTrialPosition(root.Position) then
                    local d = (root.Position - my.Position).Magnitude
                    if d < bestDist then best, bestRoot, bestHum, bestDist = plr, root, hum, d end
                end
            end
        end
        return best, bestRoot, bestHum
    end

    end

    local function MoveTowardTrialParts(mode)
        local loc = GetTrialLocation()
        local hrp = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if not hrp then return false end

        local _, race = GetPlayerRaceV4()

        if race == "Rabbit" then
            local startPoint = FindNamedPart(loc, {"StartPoint"}, 2500)
            if startPoint then
                topos(startPoint.CFrame * CFrame.new(0, 3, 0))
                return true
            end
            for _, v in ipairs(workspace:GetDescendants()) do
                if v.Name == "StartPoint" and v:IsA("BasePart") then
                    topos(v.CFrame * CFrame.new(0, 3, 0))
                    return true
                end
            end
        end

        if race == "Angel" then
            local model = workspace:FindFirstChild("Map") and workspace.Map:FindFirstChild("SkyTrial")
            model = model and model:FindFirstChild("Model")
            if model then
                local part = model:FindFirstChild("snowisland_Cylinder.081", true)
                if part and part:IsA("BasePart") then
                    topos(part.CFrame * CFrame.new(0, 3, 0))
                    return true
                end
            end
        end

        if not loc then return false end

        local candidates = {}
        for _, v in ipairs(loc:GetDescendants()) do
            if v:IsA("BasePart") then
                local d = (v.Position - hrp.Position).Magnitude
                if d > 6 and d < 1800 then
                    local n = string.lower(v.Name)
                    if not n:find("wall",1,true) and not n:find("kill",1,true) then
                        candidates[#candidates+1] = v
                    end
                end
            end
        end
        if #candidates == 0 then return false end

        local target
        if mode == "up" then
            local bestScore = -math.huge
            for _, v in ipairs(candidates) do
                local dy = v.Position.Y - hrp.Position.Y
                local d = (v.Position - hrp.Position).Magnitude
                local score = dy * 3 - d * 0.15
                if score > bestScore then bestScore, target = score, v end
            end
        else
            local bestScore = math.huge
            for _, v in ipairs(candidates) do
                local d = (v.Position - hrp.Position).Magnitude
                if d < bestScore then bestScore, target = d, v end
            end
        end
        if target then
            topos(target.CFrame * CFrame.new(0, 3, 0))
            return true
        end
        return false
    end

    local function toposMirage(Pos)
        if not Pos then return end
        local char = player.Character
        local HRP = char and char:FindFirstChild("HumanoidRootPart")
        if not HRP then return end

        CancelMirageTween()

        local MIRAGE_SPEED = 310

        local startPos = HRP.Position
        local endPos   = Pos.Position

        local dx = endPos.X - startPos.X
        local dz = endPos.Z - startPos.Z
        local horizDist = math.sqrt(dx*dx + dz*dz)

        if horizDist > 30 then
            local travelY = math.max(startPos.Y, endPos.Y)
            local midPos  = CFrame.new(endPos.X, travelY, endPos.Z)
            local d1 = (midPos.Position - startPos).Magnitude
            local dur1 = math.max(d1 / MIRAGE_SPEED, 0.05)

            currentMirageTween = TweenService:Create(
                HRP,
                TweenInfo.new(dur1, Enum.EasingStyle.Linear),
                {CFrame = midPos}
            )
            currentMirageTween:Play()
            currentMirageTween.Completed:Connect(function()
                currentMirageTween = nil
                local d2 = math.abs(travelY - endPos.Y)
                if d2 > 1 then
                    local dur2 = math.max(d2 / MIRAGE_SPEED, 0.05)
                    currentMirageTween = TweenService:Create(
                        HRP,
                        TweenInfo.new(dur2, Enum.EasingStyle.Linear),
                        {CFrame = CFrame.new(endPos)}
                    )
                    currentMirageTween:Play()
                end
            end)
        else
            local d = (endPos - startPos).Magnitude
            local dur = math.max(d / MIRAGE_SPEED, 0.05)
            currentMirageTween = TweenService:Create(
                HRP,
                TweenInfo.new(dur, Enum.EasingStyle.Linear),
                {CFrame = CFrame.new(endPos)}
            )
            currentMirageTween:Play()
        end
    end

    local function GetBoatPartsCached(boat)
        if boatPartsModel == boat and boat and boat.Parent then
            return boatPartsCache
        end

        if boatCacheConn then
            pcall(function() boatCacheConn:Disconnect() end)
            boatCacheConn = nil
        end
        table.clear(boatPartsCache)
        table.clear(boatPartsSet)
        boatPartsModel = boat
        if not boat then return boatPartsCache end

        for _, part in ipairs(boat:GetDescendants()) do
            if part:IsA("BasePart") then
                boatPartsCache[#boatPartsCache + 1] = part
                boatPartsSet[part] = true
            end
        end
        boatCacheConn = boat.DescendantAdded:Connect(function(part)
            if part:IsA("BasePart") and not boatPartsSet[part] then
                boatPartsSet[part] = true
                boatPartsCache[#boatPartsCache + 1] = part
            end
        end)
        return boatPartsCache
    end

    local function GetMyBoat()
        local char = player.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum and hum.SeatPart then
                local b = hum.SeatPart.Parent
                if b and b:IsA("Model") then

        if not boats then return nil end
        for _, b in ipairs(boats:GetChildren()) do
            if b:IsA("Model") then
                local owner = b:FindFirstChild("Owner")
                if owner and owner.Value == player.Name then
                    local hp = b:FindFirstChild("Health") or b:FindFirstChild("Humanoid")
                    if hp then
                        local h = hp:IsA("Humanoid") and hp.Health or hp.Value
                        if h and h > 0 then
                            return b
                        end
                    else
                        return b
                    end
                end
            end
        end
        return nil
    end

        if not boat then return nil end
        return boat.PrimaryPart
            or boat:FindFirstChild("VehicleSeat", true)
            or boat:FindFirstChildWhichIsA("BasePart")

    local function NoclipBoat(boat)
        if not boat then return end
        local parts = GetBoatPartsCached(boat)
        for i = 1, #parts do
            local p = parts[i]
            if p and p.Parent and p.CanCollide then
                p.CanCollide = false
            end

    end

    local function TakeOwnership(boat)
        if not boat or boatOwnershipModel == boat then return end
        boatOwnershipModel = boat
        local parts = GetBoatPartsCached(boat)
        for i = 1, #parts do
            local p = parts[i]
            if p and p.Parent then
                pcall(function() p:SetNetworkOwner(player) end)
            end
        end
    end

        if (tick() - lastBuyTime) < 5 then return end

    local function SitInSeat(seat)
        if not seat or isSitting then return end
        isSitting = true

                task.wait(0.3)

        end

        isSitting = false
    end

    local function MoveBoat(boat, targetPos, dt)
        if not boat or not boat.Parent then return false end

        local root = GetRootPart(boat)
        if not root then return false end

        local currentPos = root.Position
        local delta = targetPos - currentPos
        local dx, dy, dz = delta.X, delta.Y, delta.Z
        local distSq = dx * dx + dy * dy + dz * dz
        local dist = math.sqrt(distSq)

        if dist < 80 then return true end

        local speed = math.clamp(_G.BoatSpeed or 100, 1, 100) * 35
        local maxStep = speed * dt

        if dist > maxStep then
            local inv = maxStep / dist
            dx, dy, dz = dx * inv, dy * inv, dz * inv
        end

        delta = Vector3.new(dx, 0, dz)

        for _, p in ipairs(GetBoatPartsCached(boat)) do
            if p.Parent then
                pcall(function()
                    p.CFrame = p.CFrame + delta
                end)
            end
        end

        return false
    end

                task.wait(1)

            if not currentBoat then
                print("🚤 مفيش قارب — بنشتري...")
                BuyBoat()
                task.wait(4)
                continue
            end

            local seat = GetSeat(currentBoat)
            local char = player.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")

            if not seat then
                task.wait(0.5)
                continue
            end

            if not hum or seat.Occupant ~= hum then
                print("🚤 بنركب القارب...")
                SitInSeat(seat)
                task.wait(1)
                continue
            end

            TakeOwnership(currentBoat)
            if _G.BoatNoclip then
                NoclipBoat(currentBoat)
            end

            if not boatMoveConn then
                print("🚤 بدأنا نبحر لميراج!")
                boatMoveConn = RunService.Heartbeat:Connect(function(dt)
                    if not _G.MirageBoatMode then
                        if boatMoveConn then boatMoveConn:Disconnect() boatMoveConn = nil end
                        return
                    end
                    if not currentBoat or not currentBoat.Parent or not boatTargetPos then return end

                    local arrived = MoveBoat(currentBoat, boatTargetPos, dt)

                    if _G.BoatNoclip then
                        NoclipBoat(currentBoat)
                    end

                    if arrived then
                        print("✅ وصلنا الميراج!")
                        _G.MirageBoatMode = false
                        _G.BoatNoclip     = false
                        if boatMoveConn then boatMoveConn:Disconnect() boatMoveConn = nil end
                    end
                end)
            end
        end
    end)

local function MakeDynamicDropdown(parent, text, initialOptions, default, callback)
    local isOpen = false
    local selected = default or (initialOptions[1] or "")
    local optionsList = initialOptions

    local DropFrame = Instance.new("Frame")
    DropFrame.Parent = parent
    DropFrame.Size = UDim2.new(1, -8, 0, 34)
    DropFrame.BackgroundColor3 = Color3.fromRGB(24, 24, 28)
    DropFrame.BorderSizePixel = 0
    DropFrame.ClipsDescendants = true
    Instance.new("UICorner", DropFrame).CornerRadius = UDim.new(0, 5)
    local DS = Instance.new("UIStroke", DropFrame)
    DS.Color = Color3.fromRGB(45, 45, 50)

    local DropButton = Instance.new("TextButton")
    DropButton.Parent = DropFrame
    DropButton.Size = UDim2.new(1, 0, 0, 34)
    DropButton.BackgroundTransparency = 1
    DropButton.Text = "  " .. text .. " : " .. selected
    DropButton.TextColor3 = Color3.fromRGB(255, 255, 255)
    DropButton.Font = Enum.Font.ArialBold
    DropButton.TextSize = 11
    DropButton.TextXAlignment = Enum.TextXAlignment.Left

    local Arrow = Instance.new("TextLabel")
    Arrow.Parent = DropButton
    Arrow.Size = UDim2.new(0, 20, 1, 0)
    Arrow.Position = UDim2.new(1, -25, 0, 0)
    Arrow.BackgroundTransparency = 1
    Arrow.Text = "▼"
    Arrow.TextColor3 = Color3.fromRGB(180, 180, 180)
    Arrow.Font = Enum.Font.ArialBold
    Arrow.TextSize = 10

    local OptionsHolder = Instance.new("Frame")
    OptionsHolder.Parent = DropFrame
    OptionsHolder.Position = UDim2.new(0, 0, 0, 34)
    OptionsHolder.Size = UDim2.new(1, 0, 0, 0)
    OptionsHolder.BackgroundTransparency = 1
    local OL = Instance.new("UIListLayout", OptionsHolder)
    OL.SortOrder = Enum.SortOrder.LayoutOrder
    OL.Padding = UDim.new(0, 2)

    local Api = { Selected = selected }

    local function rebuild()
        for _, c in ipairs(OptionsHolder:GetChildren()) do
            if c:IsA("TextButton") then c:Destroy() end
        end
        OptionsHolder.Size = UDim2.new(1, 0, 0, #optionsList * 26)
        for _, opt in ipairs(optionsList) do
            local OptBtn = Instance.new("TextButton")
            OptBtn.Parent = OptionsHolder
            OptBtn.Size = UDim2.new(1, 0, 0, 24)
            OptBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 36)
            OptBtn.BorderSizePixel = 0
            OptBtn.Text = "  " .. opt
            OptBtn.TextColor3 = Color3.fromRGB(220, 220, 220)
            OptBtn.Font = Enum.Font.ArialBold
            OptBtn.TextSize = 11
            OptBtn.TextXAlignment = Enum.TextXAlignment.Left
            Instance.new("UICorner", OptBtn).CornerRadius = UDim.new(0, 4)
            OptBtn.MouseButton1Click:Connect(function()
                Api.Selected = opt
                DropButton.Text = "  " .. text .. " : " .. opt
                isOpen = false
                TweenService:Create(DropFrame, TweenInfo.new(0.2), {Size = UDim2.new(1, -8, 0, 34)}):Play()
                Arrow.Text = "▼"
                pcall(callback, opt)
            end)
        end
    end

    DropButton.MouseButton1Click:Connect(function()
        isOpen = not isOpen
        if isOpen then
            TweenService:Create(DropFrame, TweenInfo.new(0.2), {Size = UDim2.new(1, -8, 0, 34 + #optionsList * 26)}):Play()
            Arrow.Text = "▲"
        else
            TweenService:Create(DropFrame, TweenInfo.new(0.2), {Size = UDim2.new(1, -8, 0, 34)}):Play()
            Arrow.Text = "▼"
        end
    end)

    function Api:SetOptions(newOptions, newSelected)
        optionsList = newOptions or {}
        Api.Selected = newSelected or newOptions[1]
        DropButton.Text = "  " .. text .. " : " .. Api.Selected
        rebuild()
        if isOpen then
            TweenService:Create(DropFrame, TweenInfo.new(0.2), {Size = UDim2.new(1, -8, 0, 34 + #optionsList * 26)}):Play()
        end
    end

    rebuild()
    return Api
end

local isIslandTeleporting = false

local ToggleTeleport = nil

local function CleanupForces()
    if antiGravForce then antiGravForce:Destroy() antiGravForce = nil end
    if freezeVel then freezeVel:Destroy() freezeVel = nil end
end

local function ApplyAntiGravity(HRP)
    antiGravForce = Instance.new("BodyForce")
    antiGravForce.Name = "IslandAntiGrav"
    antiGravForce.Force = Vector3.new(0, HRP:GetMass() * workspace.Gravity, 0)
    antiGravForce.Parent = HRP
end

local function ApplyFreeze(HRP)
    if freezeVel then freezeVel:Destroy() end
    freezeVel = Instance.new("BodyVelocity")
    freezeVel.Name = "IslandFreeze"
    freezeVel.MaxForce = Vector3.new(1e5, 1e5, 1e5)
    freezeVel.Velocity = Vector3.new(0, 0, 0)
    freezeVel.Parent = HRP
end

local function StopTeleport(freezeInPlace)
    if currentIslandTween then
        pcall(function() currentIslandTween:Cancel() end)
        currentIslandTween = nil
    end
    isIslandTeleporting = false

    local char = player.Character
    local HRP = char and char:FindFirstChild("HumanoidRootPart")

    if freezeInPlace and HRP then
        if antiGravForce then antiGravForce:Destroy() antiGravForce = nil end
        ApplyFreeze(HRP)
    else
        CleanupForces()
    end
end

local function StartTeleport()
    local char = player.Character
    local HRP = char and char:FindFirstChild("HumanoidRootPart")
    if not HRP then return false end
    if not SelectedIslandCFrame then return false end

    CleanupForces()
    if currentIslandTween then pcall(function() currentIslandTween:Cancel() end) end
    if currentTween then pcall(function() currentTween:Cancel() end) end

    isIslandTeleporting = true
    ApplyAntiGravity(HRP)
    ApplyFreeze(HRP)

    local SPEED = 200

    local function DoneArrive()
        if not isIslandTeleporting then return end
        isIslandTeleporting = false
        CleanupForces()
        if ToggleTeleport and ToggleTeleport.SetOff then
            pcall(ToggleTeleport.SetOff)
        end
    end

    if SelectedIslandWaypoint then
        pcall(function()
            HRP.CFrame = SelectedIslandWaypoint
        end)
        task.wait(0.15)
    end

    local startPos = HRP.Position
    local endPos = SelectedIslandCFrame.Position

    local dx = endPos.X - startPos.X
    local dz = endPos.Z - startPos.Z
    local horizDist = math.sqrt(dx*dx + dz*dz)

    if horizDist > 30 then
        local travelY = math.max(startPos.Y, endPos.Y)
        local midPos = CFrame.new(endPos.X, travelY, endPos.Z)
        local d1 = (midPos.Position - startPos).Magnitude
        local dur1 = math.max(d1 / SPEED, 0.05)

        currentIslandTween = TweenService:Create(HRP, TweenInfo.new(dur1, Enum.EasingStyle.Linear), {CFrame = midPos})
        currentIslandTween:Play()
        currentIslandTween.Completed:Connect(function()
            if not isIslandTeleporting then return end
            local d2 = math.abs(travelY - endPos.Y)
            if d2 > 1 then
                local dur2 = math.max(d2 / SPEED, 0.05)
                currentIslandTween = TweenService:Create(HRP, TweenInfo.new(dur2, Enum.EasingStyle.Linear), {CFrame = CFrame.new(endPos)})
                currentIslandTween:Play()
                currentIslandTween.Completed:Connect(DoneArrive)
            else
                DoneArrive()
            end
        end)
    else
        local d = (endPos - startPos).Magnitude
        local dur = math.max(d / SPEED, 0.05)
        currentIslandTween = TweenService:Create(HRP, TweenInfo.new(dur, Enum.EasingStyle.Linear), {CFrame = CFrame.new(endPos)})
        currentIslandTween:Play()
        currentIslandTween.Completed:Connect(DoneArrive)
    end
    return true
end

local function MakeToggleLikeButton(parent, text, onTurnedOn, onTurnedOff)
    local state = false

    local ToggleFrame = Instance.new("TextButton")
    ToggleFrame.Parent = parent
    ToggleFrame.Size = UDim2.new(1, -8, 0, 40)
    ToggleFrame.BackgroundColor3 = Color3.fromRGB(24, 24, 28)
    ToggleFrame.BorderSizePixel = 0
    ToggleFrame.Text = "  " .. text
    ToggleFrame.TextColor3 = Color3.fromRGB(255, 255, 255)
    ToggleFrame.Font = Enum.Font.ArialBold
    ToggleFrame.TextSize = 12
    ToggleFrame.TextXAlignment = Enum.TextXAlignment.Left
    ToggleFrame.AutoButtonColor = false
    Instance.new("UICorner", ToggleFrame).CornerRadius = UDim.new(0, 5)
    local Stroke = Instance.new("UIStroke", ToggleFrame)
    Stroke.Color = Color3.fromRGB(45, 45, 50)

    local Track = Instance.new("Frame")
    Track.Parent = ToggleFrame
    Track.Size = UDim2.new(0, 38, 0, 20)
    Track.Position = UDim2.new(1, -46, 0.5, -10)
    Track.BackgroundColor3 = Color3.fromRGB(60, 60, 65)
    Track.BorderSizePixel = 0
    Instance.new("UICorner", Track).CornerRadius = UDim.new(0.5, 0)

    local Knob = Instance.new("Frame")
    Knob.Parent = Track
    Knob.Size = UDim2.new(0, 14, 0, 14)
    Knob.Position = UDim2.new(0, 2, 0.5, -7)
    Knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    Knob.BorderSizePixel = 0
    Instance.new("UICorner", Knob).CornerRadius = UDim.new(0.5, 0)

    local function AnimateToggle(newState)
        local targetPosition = newState and UDim2.new(1, -16, 0.5, -7) or UDim2.new(0, 2, 0.5, -7)
        local targetColor = newState and Color3.fromRGB(46, 204, 113) or Color3.fromRGB(60, 60, 65)
        local ti = TweenInfo.new(0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
        TweenService:Create(Knob, ti, {Position = targetPosition}):Play()
        TweenService:Create(Track, ti, {BackgroundColor3 = targetColor}):Play()
    end

    local Api = {}
    Api.State = false
    Api.SetOn = function()
        state = true
        Api.State = true
        AnimateToggle(true)
    end
    Api.SetOff = function()
        state = false
        Api.State = false
        AnimateToggle(false)
    end

    ToggleFrame.MouseButton1Click:Connect(function()
        if state then
            Api.SetOff()
            if onTurnedOff then pcall(onTurnedOff) end
        else
            local ok = onTurnedOn and onTurnedOn() or true
            if ok then
                Api.SetOn()
            end
        end
    end)

    return Api
end
