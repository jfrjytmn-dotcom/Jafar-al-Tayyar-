-- ============================================================
-- 📢 شاشة "تحت التعديل"
-- ============================================================
local Players = game:GetService("Players")
local TweenService = game:GetService("TweenService")
local LocalPlayer = Players.LocalPlayer

-- احذف النسخة القديمة لو موجودة
local existing = LocalPlayer:WaitForChild("PlayerGui"):FindFirstChild("UnderMaintenanceGUI")
if existing then existing:Destroy() end

local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "UnderMaintenanceGUI"
ScreenGui.ResetOnSpawn = false
ScreenGui.IgnoreGuiInset = true
ScreenGui.DisplayOrder = 999
ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
ScreenGui.Parent = LocalPlayer:WaitForChild("PlayerGui")

-- الخلفية (معتمة قليلاً)
local Backdrop = Instance.new("Frame")
Backdrop.Size = UDim2.new(1, 0, 1, 0)
Backdrop.BackgroundColor3 = Color3.fromRGB(0, 0, 0)
Backdrop.BackgroundTransparency = 1
Backdrop.BorderSizePixel = 0
Backdrop.Parent = ScreenGui

-- الإطار الرئيسي الوسط
local MainFrame = Instance.new("Frame")
MainFrame.AnchorPoint = Vector2.new(0.5, 0.5)
MainFrame.Position = UDim2.new(0.5, 0, 0.5, 0)
MainFrame.Size = UDim2.new(0, 500, 0, 160)
MainFrame.BackgroundColor3 = Color3.fromRGB(12, 12, 16)
MainFrame.BackgroundTransparency = 0.15
MainFrame.BorderSizePixel = 0
MainFrame.Parent = ScreenGui

local Corner = Instance.new("UICorner", MainFrame)
Corner.CornerRadius = UDim.new(0, 14)

local Stroke = Instance.new("UIStroke", MainFrame)
Stroke.Color = Color3.fromRGB(255, 180, 0)
Stroke.Thickness = 2
Stroke.Transparency = 0.1

-- تأثير التوهج للحدود
task.spawn(function()
    while MainFrame.Parent do
        TweenService:Create(Stroke, TweenInfo.new(1.2, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {
            Transparency = 0.7
        }):Play()
        task.wait(1.2)
        TweenService:Create(Stroke, TweenInfo.new(1.2, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {
            Transparency = 0.1
        }):Play()
        task.wait(1.2)
    end
end)

-- الأيقونة (تحذير)
local Icon = Instance.new("TextLabel")
Icon.Size = UDim2.new(0, 60, 0, 60)
Icon.Position = UDim2.new(0.5, 0, 0, -30)
Icon.AnchorPoint = Vector2.new(0.5, 0.5)
Icon.BackgroundColor3 = Color3.fromRGB(255, 180, 0)
Icon.BorderSizePixel = 0
Icon.Text = "⚠️"
Icon.TextSize = 32
Icon.Font = Enum.Font.GothamBold
Icon.Parent = MainFrame

local IconCorner = Instance.new("UICorner", Icon)
IconCorner.CornerRadius = UDim.new(0.5, 0)

-- النص الرئيسي
local Title = Instance.new("TextLabel")
Title.Size = UDim2.new(1, -40, 0, 60)
Title.Position = UDim2.new(0, 20, 0, 55)
Title.BackgroundTransparency = 1
Title.Text = "السيكريبت تحت التعديل"
Title.TextColor3 = Color3.fromRGB(255, 255, 255)
Title.TextSize = 30
Title.Font = Enum.Font.GothamBold
Title.TextStrokeTransparency = 0.4
Title.TextStrokeColor3 = Color3.fromRGB(255, 180, 0)
Title.Parent = MainFrame

-- النص الفرعي
local Subtitle = Instance.new("TextLabel")
Subtitle.Size = UDim2.new(1, -40, 0, 30)
Subtitle.Position = UDim2.new(0, 20, 0, 115)
Subtitle.BackgroundTransparency = 1
Subtitle.Text = "⏳ يرجى الانتظار حتى يتم الانتهاء..."
Subtitle.TextColor3 = Color3.fromRGB(180, 180, 190)
Subtitle.TextSize = 15
Subtitle.Font = Enum.Font.Gotham
Subtitle.Parent = MainFrame

-- تأثير نبض على النص الرئيسي
task.spawn(function()
    while Title.Parent do
        TweenService:Create(Title, TweenInfo.new(1, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {
            TextSize = 34
        }):Play()
        task.wait(1)
        TweenService:Create(Title, TweenInfo.new(1, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {
            TextSize = 30
        }):Play()
        task.wait(1)
    end
end)

-- تأثير نبض على الأيقونة
task.spawn(function()
    while Icon.Parent do
        TweenService:Create(Icon, TweenInfo.new(0.8, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {
            Rotation = 10
        }):Play()
        task.wait(0.8)
        TweenService:Create(Icon, TweenInfo.new(0.8, Enum.EasingStyle.Sine, Enum.EasingDirection.InOut), {
            Rotation = -10
        }):Play()
        task.wait(0.8)
    end
end)

-- دخول ناعم
MainFrame.Size = UDim2.new(0, 0, 0, 0)
TweenService:Create(MainFrame, TweenInfo.new(0.6, Enum.EasingStyle.Back, Enum.EasingDirection.Out), {
    Size = UDim2.new(0, 500, 0, 160)
}):Play()

print("✅ شاشة 'تحت التعديل' ظهرت")
